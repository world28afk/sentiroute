    ```typescript
    import { existsSync } from 'node:fs';
    import { join } from 'node:path';
    import { homedir, platform } from 'node:os';

    export function resolveConfigPath(): string {
      const envPath = process.env['SENTIROUTE_CONFIG'];
      if (envPath && existsSync(envPath)) return envPath;

      const cwdPath = join(process.cwd(), 'sentiroute.yaml');
      if (existsSync(cwdPath)) return cwdPath;

      const cwdYmlPath = join(process.cwd(), 'sentiroute.yml');
      if (existsSync(cwdYmlPath)) return cwdYmlPath;

      const userDir = platform() === 'win32'
        ? join(process.env['APPDATA'] || join(homedir(), 'AppData', 'Roaming'), 'SentiRoute')
        : join(homedir(), '.config', 'sentiroute');

      const userPath = join(userDir, 'config.yaml');
      if (existsSync(userPath)) return userPath;

      throw new Error(
        `No config file found. Create a sentiroute.yaml file or set SENTIROUTE_CONFIG.\n` +
        `Searched:\n` +
        `  - ${envPath ? envPath : '(SENTIROUTE_CONFIG not set or file not found)'}\n` +
        `  - ${cwdPath}\n` +
        `  - ${cwdYmlPath}\n` +
        `  - ${userPath}`
      );
    }
    ```

  <action>
    Create src/config/loader.ts. Implement the full config loading pipeline as documented in RESEARCH.md (lines 160-230):

    Pipeline: readFileSync → parseDocument() → doc.toJS() → configSchema.safeParse() → map Zod errors back to YAML lines via doc.getIn(issue.path, true)

    **Key implementation details:**

    1. Use `yaml.parseDocument()` (NOT `yaml.parse()`) — parseDocument returns a Document with source-position tracking on AST nodes. This is required for mapping Zod errors back to line numbers.

    2. YAML syntax errors: catch `YAMLParseError` from parseDocument, extract `linePos[0].line` and `linePos[0].col` from the error object, wrap in ConfigValidationError.

    3. YAML warnings: check `doc.warnings` array after parsing. For each warning, `console.warn()` with the file path and line number. Do NOT fail validation on warnings.

    4. Zod validation: use `configSchema.safeParse(data)` (not .parse()). On failure, iterate `result.error.issues` and for each issue:
       - Build path string from `issue.path.join('.')`
       - Use `doc.getIn(issue.path, true)` to get the YAML AST node with source position
       - Extract `node?.linePos?.[0]?.line` and `node?.linePos?.[0]?.col`
       - Create ValidationIssue with message, filePath, line, column, expected, received
       - For expected/received: parse issue.message for "Expected X, received Y" patterns

    5. Return `result.data` as typed Config on success.

    6. Read RESEARCH.md lines 168-230 for the exact implementation pattern. The code in that section is the reference implementation — implement it faithfully.

    **File read error handling:** Wrap readFileSync in try/catch. If file not found or unreadable, throw ConfigValidationError with a single ValidationIssue containing the OS error message and the filePath.

    **Important:** The Config object returned has an `_configPath` field (from schema.ts's `.optional()` field) — set it to the resolved configPath before returning so downstream code (health endpoint) can display which config file was loaded.
