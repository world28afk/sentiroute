```typescript
// ── Static file serving for dashboard ──
// Resolve dashboard directory: in dev, use source dashboard/; in production (dist), use dist/dashboard/
const __dirname = dirname(fileURLToPath(import.meta.url));
const isProduction = __dirname.includes('/dist/') || __dirname.includes('\\dist\\');
const dashboardRoot = isProduction
  ? join(__dirname, '../dashboard')       // dist/server/ -> dist/dashboard/
  : join(__dirname, '../../dashboard');    // src/server/ -> dashboard/

await app.register(FastifyStatic, {
  root: dashboardRoot,
  prefix: '/dashboard/',
  wildcard: false,           // Do NOT serve /dashboard/* as index.html — explicit routes only
  index: ['index.html'],     // /dashboard/ → index.html
  maxAge: '1h',
  immutable: false,
});

// ── Dashboard API routes ──
await app.register(dashboardApi, { configManager, sentimentState });
```

**Change the routeOpts for proxy routes:**

The existing `messages.ts` and `chat.ts` routes read `opts.config` (a `Config` object). Since we're now passing `ConfigManager` instead of `Config`, update how proxy routes receive config:

The proxy routes (`messagesRoute` and `chatRoute`) still need a `Config` object. The cleanest approach: pass `configManager.config` as a separate `config` property alongside `configManager`:

```typescript
// In createApp(), after creating the Fastify instance:
const liveConfig = configManager.config; // initial snapshot
const dataDir = liveConfig._configPath ? dirname(liveConfig._configPath) : process.cwd();

// Proxy routes continue to receive { config, dataDir, sentimentState }
// where config = configManager.config (current in-memory state)
const routeOpts = { config: liveConfig, dataDir, sentimentState };

app.register(healthRoute, { config: liveConfig });
app.register(messagesRoute, routeOpts);
app.register(chatRoute, routeOpts);

// BUT: proxy routes read config per-request, and configManager.config may have been
// updated by dashboard. To handle this, the proxy routes should read from
// configManager.config directly. The simplest fix: expose configManager as a
// property on the opts object, OR update the proxy routes to use configManager.

// For NOW: proxy routes still receive { config, dataDir, sentimentState }.
// The config is the snapshot at startup. Runtime parameter hot-updates via
// ConfigManager.updateSentiment() modify configManager.config.sentiment in-place,
// which IS the same object reference as liveConfig.sentiment (since SentimentConfig
// is an object, not a primitive). So sentiment hot-updates work for proxy routes.
// Structural changes (model_slots) replace the entire config object and require restart.
```

**IMPORTANT design note for the executor:**

The proxy routes (`messages.ts`, `chat.ts`) currently destructure `opts.config` and read `opts.config.model_slots` and `opts.config.sentiment` at request time. Since `ConfigManager.updateSentiment()` mutates `this.config.sentiment` in-place (using object spread on the `sentiment` sub-object), and the `liveConfig` reference is the same object as `configManager.config`, sentiment hot-updates WILL be visible to proxy routes without any changes to `messages.ts` or `chat.ts`.

Structural changes (model_slots replace) are different — `configManager.config = result.data` replaces the entire object. For these, the restart banner is shown. This matches the research decision: "Runtime params hot-update. Structural changes write to YAML + display restart banner."

DO NOT modify `messages.ts` or `chat.ts` in this task. The existing proxy routes work correctly with runtime parameter hot-updates because they read from `opts.config.sentiment` at request time, and `opts.config` shares the same `sentiment` object reference as `configManager.config`.

```javascript
    await this.fetchConfig();
  },

  async fetchConfig() {
    this.loading = true;
    try {
      const res = await fetch('/api/dashboard/config');
      if (!res.ok) throw new Error('Failed to load config');
      this.configData = await res.json();
    } catch (e) {
      this.error = e.message;
    } finally {
      this.loading = false;
    }
  },

  async saveRuntimeParams() {
    try {
      const body = {
        sentiment: {
          threshold: Number(this.configData.sentiment.threshold),
          decayRate: Number(this.configData.sentiment.decayRate),
          cooldownMs: Number(this.configData.sentiment.cooldownMs),
          antiFlapMs: Number(this.configData.sentiment.antiFlapMs),
          weights: { ...this.configData.sentiment.weights },
        },
      };
      const res = await fetch('/api/dashboard/config', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
      });
      const result = await res.json();
      if (res.ok) {
        this.saveMessage = result.message;
        this.restartRecommended = result.restartRecommended;
      } else {
        this.saveError = result.error || 'Save failed';
      }
    } catch (e) {
      this.saveError = e.message;
    }
  },

  async saveFullConfig() {
    try {
      const res = await fetch('/api/dashboard/config', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ model_slots: this.configData.model_slots }),
      });
      const result = await res.json();
      if (res.ok) {
        this.saveMessage = result.message;
        this.restartRecommended = result.restartRecommended;
        // Re-fetch to get sanitized config back
        await this.fetchConfig();
      } else {
        this.saveError = result.error || 'Save failed';
      }
    } catch (e) {
      this.saveError = e.message;
    }
  },

  // Add a new upstream to a slot
  addUpstream(slotId) {
    const slot = this.configData.model_slots[slotId];
    if (!slot) return;
    slot.upstreams.push({
      name: '',
      endpoint: 'https://api.example.com/v1',
      api_key: '',
      upstream_model: '',
      format: 'anthropic',
      timeoutMs: 120000,
    });
  },

  // Remove an upstream from a slot
  removeUpstream(slotId, index) {
    const slot = this.configData.model_slots[slotId];
    if (!slot || slot.upstreams.length <= 1) return;
    slot.upstreams.splice(index, 1);
  },

    if (key.length <= 8) return '***';
    return key.slice(0, 2) + '...' + key.slice(-6);
```

```javascript
      if (!res.ok) throw new Error('Failed to load state');
      const data = await res.json();
      this.slots = data.slots;
      this.threshold = data.threshold;
      this.now = data.now;
      this.lastUpdated = new Date().toLocaleTimeString();
      this.error = '';
    } catch (e) {
      this.error = e.message;
    }
  },

  formatCooldown(until) {
    if (!until) return '—';
    const remaining = Math.max(0, until - Date.now());
    if (remaining <= 0) return 'none';
    const s = Math.floor(remaining / 1000);
    if (s < 60) return s + 's';
    const m = Math.floor(s / 60);
    if (m < 60) return m + 'm ' + (s % 60) + 's';
    const h = Math.floor(m / 60);
    return h + 'h ' + (m % 60) + 'm';
  },

  getScoreClass(score) {
    return score > this.threshold ? 'danger' : 'safe';
  },
}));
```
